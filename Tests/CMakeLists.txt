find_package(Threads REQUIRED)
add_executable(stutter_lab_core_tests CoreTests.cpp)
target_include_directories(stutter_lab_core_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../Source/Core")
target_link_libraries(stutter_lab_core_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(stutter_lab_core_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(stutter_lab_core_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME stutter_lab_core_tests COMMAND stutter_lab_core_tests)
